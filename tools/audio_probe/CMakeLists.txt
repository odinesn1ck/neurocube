cmake_minimum_required(VERSION 3.21)
set(PICO_BOARD pico2)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(audio_probe C CXX ASM)
pico_sdk_init()
set(VENDOR "$ENV{WAVESHARE_AUDIO_ROOT}" CACHE PATH "Pinned Waveshare examples/C/02_ES8311 directory")
if(NOT EXISTS "${VENDOR}/lib/es8311/es8311.c")
  message(FATAL_ERROR "Set WAVESHARE_AUDIO_ROOT to Waveshare 51a7fe28 examples/C/02_ES8311")
endif()
add_executable(audio_probe main.c ${VENDOR}/lib/es8311/es8311.c)
target_include_directories(audio_probe PRIVATE ${VENDOR}/lib/Config ${VENDOR}/lib/audio_pio ${VENDOR}/lib/es8311)
pico_generate_pio_header(audio_probe ${VENDOR}/lib/audio_pio/audio_pio.pio)
target_link_libraries(audio_probe pico_stdlib hardware_i2c hardware_pio hardware_clocks hardware_watchdog hardware_spi hardware_pwm hardware_adc hardware_dma m)
pico_enable_stdio_usb(audio_probe 1)
pico_enable_stdio_uart(audio_probe 0)
pico_set_binary_type(audio_probe no_flash)
pico_add_extra_outputs(audio_probe)
